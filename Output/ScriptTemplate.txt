public static class DicExtension
{
  /// <summary>
  /// 安全获取字典值 - 扩展方法
  /// </summary>
  public static T GetValue<T>(this Dictionary<string, object> dict, string key)
  {
    if (dict != null && dict.ContainsKey(key))
    {
      try
      {
        var value = dict[key];
        return (T) Convert.ChangeType(value, typeof(T));
      }
      catch
      {
        return default(T);
      }
    }
    return default(T);
  }

  /// <summary>
  /// 获取String
  /// </summary>
  public static string GetString(this Dictionary<string, object> dict, string key)
  {
    return dict.GetValue<string>(key);
  }

  /// <summary>
  /// 获取Int
  /// </summary>
  public static int GetInt(this Dictionary<string, object> dict, string key)
  {
    return dict.GetValue<int>(key);
  }

  /// <summary>
  /// 获取Double
  /// </summary>
  public static double GetDouble(this Dictionary<string, object> dict, string key)
  {
    return dict.GetValue<double>(key);
  }

  /// <summary>
  /// 获取Bool
  /// </summary>
  public static bool GetBool(this Dictionary<string, object> dict, string key)
  {
    return dict.GetValue<bool>(key);
  }

  /// <summary>
  /// 获取Range
  /// </summary>
  public static Tuple<double, double> GetRange(this Dictionary<string, object> dict, string key)
  {
    return dict.GetValue<Tuple<double, double>>(key);
  }
}
